  <script>
    async function generateContent() {
      const apiKey = document.getElementById('apiKey').value;
      const text = document.getElementById('inputText').value;
      const btn = document.getElementById('generateBtn');

      if (!apiKey || !text) {
        alert('Please enter both API key and input text!');
        return;
      }

      btn.innerText = 'Generating Content... ⏳';
      btn.disabled = true;

      const prompt = `Repurpose the following text into 2 distinct formats:
1. Twitter Thread (3-5 tweets with emojis).
2. Professional LinkedIn Post (engaging hook, main points, CTA).

Output format:
---TWITTER---
[Tweets here]
---LINKEDIN---
[LinkedIn post here]

Original Text:
${text}`;

      try {
        const response = await fetch('https://api.groq.com/openai/v1/chat/completions', {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${apiKey}`,
            'Content-Type': 'application/json'
          },
          body: JSON.stringify({
            model: 'llama-3.3-70b-versatile',
            messages: [{ role: 'user', content: prompt }]
          })
        });

        const data = await response.json();
        const output = data.choices[0].message.content;

        const parts = output.split('---LINKEDIN---');
        const twitterPart = parts[0].replace('---TWITTER---', '').trim();
        const linkedinPart = parts[1] ? parts[1].trim() : '';

        document.getElementById('twitterResult').innerText = twitterPart;
        document.getElementById('linkedinResult').innerText = linkedinPart;
        document.getElementById('resultSection').classList.remove('hidden');

      } catch (err) {
        alert('Error generating content. Check your API Key.');
        console.error(err);
      } finally {
        btn.innerText = '🚀 Repurpose Content';
        btn.disabled = false;
      }
    }

    function copyText(elementId) {
      const text = document.getElementById(elementId).innerText;
      navigator.clipboard.writeText(text);
      alert('Copied to clipboard!');
    }
  </script>
